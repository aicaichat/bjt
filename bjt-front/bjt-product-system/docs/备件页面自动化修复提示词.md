### 🎯 关键经验教训

#### 1. **问题诊断方法的重要性**
- **教训**：不能只看现象，必须做根本性分析
- **成功做法**：通过API测试确定问题出现在前后端哪一层
- **避免陷阱**：不要一开始就怀疑复杂的数据库或架构问题

#### 2. **调试工具的价值与清理**
- **成功做法**：创建了专门的调试工具来测试筛选逻辑，快速定位问题
- **重要原则**：修复完成后立即清理所有调试代码，保持代码整洁
- **清理内容**：
  - 移除 `frontend/src/utils/debugFilter.ts` 调试工具文件
  - 移除页面中的5个调试按钮
  - 移除全局window对象暴露的调试函数
  - 删除备件页面的backup文件

#### 3. **前后端问题定位策略**
- **成功方法**：先用curl测试API，确定后端返回的数据是否正确
- **定位流程**：API测试 → 前端数据处理检查 → 筛选逻辑验证
- **避免盲区**：不要假设前端或后端哪一层有问题，要用数据说话

#### 4. **字段验证逻辑的重要性**
- **根本问题**：后端字段验证逻辑不匹配业务规则
- **解决方案**：确保数据库设计、业务规则、前端显示三者一致
- **映射关系**：`0=不展示，1=易损，2=非易损` 在整个系统中必须统一

#### 5. **增量修复vs重构的选择**
- **成功策略**：选择增量修复，只修改有问题的核心逻辑
- **风险控制**：避免大范围重构，降低引入新bug的风险
- **测试验证**：每个小修改都立即测试验证效果

# 4. API层验证
echo -e "\n🔌 API层验证..."
echo "========================"

# 获取认证Token
echo "🔐 获取API认证Token..."
TOKEN=$(curl -s -X POST "http://localhost:8080/wp-json/bjt/v1/auth/login" \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"password123"}' | jq -r '.token')

if [ "$TOKEN" = "null" ] || [ -z "$TOKEN" ]; then
  echo "❌ 获取Token失败，请检查认证服务"
  exit 1
else
  echo "✅ Token获取成功"
fi

# 验证备件列表API
echo -e "\n📋 验证备件列表API..."
SPARE_PARTS_RESPONSE=$(curl -s "http://localhost:8080/wp-json/bjt/v1/spare-parts?exclude_hidden=false")
SPARE_PARTS_COUNT=$(echo "$SPARE_PARTS_RESPONSE" | jq '. | length' 2>/dev/null || echo "0")

if [ "$SPARE_PARTS_COUNT" -gt 0 ]; then
  echo "✅ 备件列表API返回 $SPARE_PARTS_COUNT 条数据"
  
  # 验证必需字段存在性
  echo -e "\n🔍 验证API返回字段完整性..."
  FIRST_ITEM=$(echo "$SPARE_PARTS_RESPONSE" | jq '.[0]' 2>/dev/null)
  
  # 检查必需字段
  REQUIRED_FIELDS=("app_model" "is_consumable" "image_url" "part_number" "name_zh" "name_en")
  for field in "${REQUIRED_FIELDS[@]}"; do
    HAS_FIELD=$(echo "$FIRST_ITEM" | jq "has(\"$field\")" 2>/dev/null)
    if [ "$HAS_FIELD" = "true" ]; then
      VALUE=$(echo "$FIRST_ITEM" | jq -r ".$field" 2>/dev/null)
      echo "✅ $field: $VALUE"
    else
      echo "❌ 缺少字段: $field"
    fi
  done
  
  # 验证数据类型
  echo -e "\n🔬 验证数据类型..."
  IS_CONSUMABLE_TYPE=$(echo "$FIRST_ITEM" | jq -r '.is_consumable | type' 2>/dev/null)
  NET_WEIGHT_TYPE=$(echo "$FIRST_ITEM" | jq -r '.net_weight_kg | type' 2>/dev/null)
  PCS_PER_BOX_TYPE=$(echo "$FIRST_ITEM" | jq -r '.pcs_per_box | type' 2>/dev/null)
  
  echo "is_consumable类型: $IS_CONSUMABLE_TYPE $([ "$IS_CONSUMABLE_TYPE" = "number" ] && echo "✅" || echo "❌")"
  echo "net_weight_kg类型: $NET_WEIGHT_TYPE $([ "$NET_WEIGHT_TYPE" = "number" ] && echo "✅" || echo "❌")"
  echo "pcs_per_box类型: $PCS_PER_BOX_TYPE $([ "$PCS_PER_BOX_TYPE" = "number" ] && echo "✅" || echo "❌")"
  
else
  echo "❌ 备件列表API返回空数据"
fi

# 验证筛选API
echo -e "\n🔍 验证筛选功能..."
FILTER_RESPONSE=$(curl -s "http://localhost:8080/wp-json/bjt/v1/spare-parts?is_consumable=1&exclude_hidden=false")
FILTER_COUNT=$(echo "$FILTER_RESPONSE" | jq '. | length' 2>/dev/null || echo "0")
echo "易损筛选结果: $FILTER_COUNT 条数据"

# 6. 数据一致性全链路验证
echo -e "\n🔄 数据一致性全链路验证..."
echo "===================================="

# 对比数据库和API的数据一致性
echo "🔍 验证数据库与API数据一致性..."

# 从数据库获取第一条备件数据
DB_FIRST_PART=$(docker exec -it dev-mysql-1 mysql -u root -proot bjt_product -e "
SELECT id, app_model, is_consumable, part_number, name_zh, name_en, net_weight_kg, pcs_per_box 
FROM wp_bjt_spare_parts 
WHERE status = 'publish' 
LIMIT 1;
" 2>/dev/null | tail -n +2)

if [ -n "$DB_FIRST_PART" ]; then
  DB_ID=$(echo "$DB_FIRST_PART" | awk '{print $1}')
  DB_IS_CONSUMABLE=$(echo "$DB_FIRST_PART" | awk '{print $3}')
  echo "数据库第一条备件: ID=$DB_ID, is_consumable=$DB_IS_CONSUMABLE"
  
  # 从API获取相同备件数据
  API_PART=$(curl -s "http://localhost:8080/wp-json/bjt/v1/spare-parts/$DB_ID")
  API_IS_CONSUMABLE=$(echo "$API_PART" | jq -r '.is_consumable' 2>/dev/null)
  
  if [ "$DB_IS_CONSUMABLE" = "$API_IS_CONSUMABLE" ]; then
    echo "✅ 数据库与API数据一致 (is_consumable: $DB_IS_CONSUMABLE)"
  else
    echo "❌ 数据不一致! 数据库: $DB_IS_CONSUMABLE, API: $API_IS_CONSUMABLE"
  fi
else
  echo "❌ 无法从数据库获取数据"
fi

# 验证字段映射完整性
echo -e "\n📋 验证字段映射完整性..."
SAMPLE_API_DATA=$(echo "$SPARE_PARTS_RESPONSE" | jq '.[0]' 2>/dev/null)

# 检查所有必需的展示字段
DISPLAY_FIELDS=(
  "app_model:适配机型"
  "is_consumable:是否易损"  
  "image_url:产品图片"
  "part_number:料号"
  "name_zh:中文名称"
  "name_en:英文名称"
  "spec:规格描述"
  "app_sn:适配序列号"
  "package_size_cm:包装尺寸cm"
  "package_size_inch:包装尺寸inch"
  "net_weight_kg:净重kg"
  "net_weight_lbs:净重lbs"
  "pcs_per_box:单箱数量"
)

echo "检查API返回的字段完整性:"
for field_info in "${DISPLAY_FIELDS[@]}"; do
  field=$(echo "$field_info" | cut -d':' -f1)
  desc=$(echo "$field_info" | cut -d':' -f2)
  
  HAS_FIELD=$(echo "$SAMPLE_API_DATA" | jq "has(\"$field\")" 2>/dev/null)
  if [ "$HAS_FIELD" = "true" ]; then
    VALUE=$(echo "$SAMPLE_API_DATA" | jq -r ".$field" 2>/dev/null)
    if [ "$VALUE" != "null" ] && [ -n "$VALUE" ]; then
      echo "✅ $desc ($field): 有数据"
    else
      echo "⚠️ $desc ($field): 字段存在但值为空"
    fi
  else
    echo "❌ $desc ($field): 字段缺失"
  fi
done

const formatPackageSize = (value: string, unit: string, options?: {
  normalize?: boolean; // 是否标准化格式
  language?: 'zh' | 'en';
}): string => {
  if (!value) return '';
  
  const { normalize = false, language = 'zh' } = options || {};
  
  // 常见尺寸格式处理
  const formats = [
    /^(\d+(?:\.\d+)?)\*(\d+(?:\.\d+)?)\*(\d+(?:\.\d+)?)$/, // 10*29*49
    /^(\d+(?:\.\d+)?)x(\d+(?:\.\d+)?)x(\d+(?:\.\d+)?)$/, // 10x29x49  
    /^(\d+(?:\.\d+)?)×(\d+(?:\.\d+)?)×(\d+(?:\.\d+)?)$/, // 10×29×49
    /^(\d+(?:\.\d+)?)\*(\d+(?:\.\d+)?)$/, // 21*30 (二维)
    /^(\d+(?:\.\d+)?)x(\d+(?:\.\d+)?)$/, // 21x30 (二维)
    /^(\d+(?:\.\d+)?)×(\d+(?:\.\d+)?)$/, // 21×30 (二维)
  ];
  
  // 检查是否匹配已知格式
  for (const format of formats) {
    if (format.test(value)) {
      const processedValue = normalize ? normalizeDimensionFormat(value) : value;
      return `${processedValue} ${unit}`;
    }
  }
  
  // 单一数值
  if (/^\d+(?:\.\d+)?$/.test(value)) {
    return `${value} ${unit}`;
  }
  
  // 其他格式直接添加单位
  return `${value} ${unit}`;
};
