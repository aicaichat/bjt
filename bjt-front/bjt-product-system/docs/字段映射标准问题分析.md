```typescript
export const validateFieldMapping = (productType: string, data: any) => {
  const mapping = getFieldMapping(productType);
  const missingFields = [];
  const requiredFields = Object.values(mapping).filter(field => field.required);
  
  requiredFields.forEach(field => {
    if (!data[field.key] || data[field.key] === '') {
      missingFields.push(field.key);
    }
  });
  
  return { isValid: missingFields.length === 0, missingFields };
};
```

```typescript
export const standardizeUnits = (data: any) => {
  // 统一lbs -> lb
  Object.keys(data).forEach(key => {
    if (typeof data[key] === 'string' && data[key].includes('lbs')) {
      data[key] = data[key].replace(/lbs/g, 'lb');
    }
  });
  return data;
};
```
