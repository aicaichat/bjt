## 第二部分：核心实施代码

### 2.1 ⭐ 智能单位制切换实现

**基于 AuthContext 的 `getPreferredUnit()` 实现**：

```typescript
// ✅ 核心实现：智能单位制切换
const { getPreferredUnit } = useAuth();
const preferredUnit = getPreferredUnit(); // 'metric' | 'imperial'

// 根据用户偏好自动选择字段
const getSmartFieldMapping = (fieldKey: string) => {
  const fieldConfig = MACHINE_FIELD_CONFIGS[fieldKey];
  if (!fieldConfig?.unitConfig) return fieldKey;
  
  const isImperial = preferredUnit === 'imperial';
  return isImperial ? fieldConfig.unitConfig.imperial : fieldConfig.unitConfig.metric;
};

// 示例：重量字段智能切换
// 公制用户: net_weight_kg → '单件净重(kg)' + '25.5'
// 英制用户: net_weight_lbs → 'Net Weight(lbs)' + '56.2'
```

### 2.2 🔧 标准化单位显示实现

**严格遵循"标题含单位，内容纯数值"原则**：

```typescript
// ✅ 正确实现：单位处理标准
const getFieldLabel = (fieldKey: string): string => {
  const targetField = getSmartFieldMapping(fieldKey);
  const labelConfig = MACHINE_FIELD_LABELS[currentLanguage];
  return labelConfig[targetField]; // 返回包含单位的标题
};

const getLocalizedValue = (machine: MachinePart, fieldKey: string): string => {
  const targetField = getSmartFieldMapping(fieldKey);
  const rawValue = machine[targetField];
  
  // 返回纯数值，不包含单位
  return rawValue ? String(rawValue) : '';
};

// 使用示例：
// getFieldLabel('net_weight') → '单件净重(kg)'
// getLocalizedValue(machine, 'net_weight') → '25.5'
```
