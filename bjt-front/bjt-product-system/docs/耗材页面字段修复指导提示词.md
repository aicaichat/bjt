基于前端代码与CSV标准字段的对比分析，耗材页面存在以下问题：

// ✅ 完整的筛选器实现
interface FilterState {
  selectedModel: string;         // 适用机型筛选
  selectedShape: string;         // 形状筛选
  selectedMaterial: string;      // 材质筛选
  thicknessRange: [number, number] | null; // 厚度范围
  widthRange: [number, number] | null;     // 膜宽范围
  lengthRange: [number, number] | null;    // 袋长范围
  unitSystem: 'metric' | 'imperial';       // 单位制
}

const ConsumablesFilters: React.FC = () => {
  const [filters, setFilters] = useState<FilterState>({
    selectedModel: 'all',
    selectedShape: 'all',
    selectedMaterial: 'all',
    thicknessRange: null,
    widthRange: null,
    lengthRange: null,
    unitSystem: 'metric'
  });
  
  return (
    <div className="consumables-filters">
      {/* 适用机型筛选 */}
      <div className="filter-group">
        <label>适用机型</label>
        <Select
          value={filters.selectedModel}
          onChange={(value) => setFilters(prev => ({...prev, selectedModel: value}))}
          options={modelOptions}
        />
      </div>
      
      {/* 形状筛选 */}
      <div className="filter-group">
        <label>形状</label>
        <Select
          value={filters.selectedShape}
          onChange={(value) => setFilters(prev => ({...prev, selectedShape: value}))}
          options={shapeOptions}
        />
      </div>
      
      {/* 材质筛选 */}
      <div className="filter-group">
        <label>材质</label>
        <Select
          value={filters.selectedMaterial}
          onChange={(value) => setFilters(prev => ({...prev, selectedMaterial: value}))}
          options={materialOptions}
        />
      </div>
      
      {/* 厚度范围筛选 */}
      <div className="filter-group">
        <label>厚度/克重 {filters.unitSystem === 'metric' ? '(um/gsm)' : '(mil/#)'}</label>
        <Slider
          range
          value={filters.thicknessRange}
          onChange={(range) => setFilters(prev => ({...prev, thicknessRange: range}))}
          min={0}
          max={filters.unitSystem === 'metric' ? 100 : 4}
          step={filters.unitSystem === 'metric' ? 1 : 0.1}
        />
      </div>
      
      {/* 膜宽范围筛选 */}
      <div className="filter-group">
        <label>膜宽 {filters.unitSystem === 'metric' ? '(cm)' : '(inch)'}</label>
        <Slider
          range
          value={filters.widthRange}
          onChange={(range) => setFilters(prev => ({...prev, widthRange: range}))}
          min={0}
          max={filters.unitSystem === 'metric' ? 200 : 80}
        />
      </div>
      
      {/* 袋长范围筛选 */}
      <div className="filter-group">
        <label>袋长 {filters.unitSystem === 'metric' ? '(cm)' : '(inch)'}</label>
        <Slider
          range
          value={filters.lengthRange}
          onChange={(range) => setFilters(prev => ({...prev, lengthRange: range}))}
          min={0}
          max={filters.unitSystem === 'metric' ? 100 : 40}
        />
      </div>
      
      {/* 单位制切换 */}
      <div className="filter-group">
        <Radio.Group
          value={filters.unitSystem}
          onChange={(e) => setFilters(prev => ({...prev, unitSystem: e.target.value}))}
        >
          <Radio value="metric">公制</Radio>
          <Radio value="imperial">英制</Radio>
        </Radio.Group>
      </div>
    </div>
  );
};
