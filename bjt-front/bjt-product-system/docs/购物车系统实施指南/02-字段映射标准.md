export const FIELD_FORMATTERS = {
  // 图片格式化器
  image: (value: string): string => {
    if (!value) return '';

  },
  
  // 重量格式化器（纯数值）
  weight: (value: number): string => {

  },

  dimension: (value: string): string => {

  },
  
  // 宽度格式化器

    if (typeof value !== 'number' || isNaN(value)) return '';
    return value.toFixed(1);
  },
