# 多语言配置方案

## 🎯 多语言支持目标

为购物车系统提供完整的中英文支持，确保字段标签、界面文案、数据内容的完全国际化，同时支持智能语言检测和用户偏好保存。

## 🔧 多语言Hook实现

### useI18n Hook
```typescript
import { useState, useEffect, useMemo } from 'react';
import { loadLanguageResources } from '../utils/i18nLoader';

interface I18nConfig {
  defaultLanguage: 'zh' | 'en';
  fallbackLanguage: 'zh' | 'en';
  supportedLanguages: ('zh' | 'en')[];
}

export const useI18n = (config: I18nConfig = {
  defaultLanguage: 'zh',
  fallbackLanguage: 'zh',
  supportedLanguages: ['zh', 'en']
}) => {
  const [currentLanguage, setCurrentLanguage] = useState<'zh' | 'en'>(config.defaultLanguage);
  const [resources, setResources] = useState<any>({});
  const [isLoading, setIsLoading] = useState(true);
  
  // 加载语言资源
  useEffect(() => {
    const loadResources = async () => {
      setIsLoading(true);
      try {
        const languageResources = await loadLanguageResources(currentLanguage);
        setResources(languageResources);
      } catch (error) {
        console.error(`加载语言资源失败 (${currentLanguage}):`, error);
        // 尝试加载备用语言
        if (currentLanguage !== config.fallbackLanguage) {
          const fallbackResources = await loadLanguageResources(config.fallbackLanguage);
          setResources(fallbackResources);
        }
      } finally {
        setIsLoading(false);
      }
    };
    
    loadResources();
  }, [currentLanguage, config.fallbackLanguage]);
  
  // 翻译函数
  const t = useMemo(() => {
    return (key: string, params?: Record<string, any>, fallback?: string): string => {
      if (isLoading || !resources) {
        return fallback || key;
      }
      
      // 支持嵌套键路径，如 'cart.common.loading'
      const keyPath = key.split('.');
      let value = resources;
      
      for (const path of keyPath) {
        if (value && typeof value === 'object' && path in value) {
          value = value[path];
        } else {
          value = null;
          break;
        }
      }
      
      let result = typeof value === 'string' ? value : (fallback || key);
      
      // 参数替换
      if (params && typeof result === 'string') {
        Object.entries(params).forEach(([paramKey, paramValue]) => {
          const placeholder = `{${paramKey}}`;
          result = result.replace(new RegExp(placeholder, 'g'), String(paramValue));
        });
      }
      
      return result;
    };
  }, [resources, isLoading]);
  
  // 切换语言
  const switchLanguage = (language: 'zh' | 'en') => {
    if (config.supportedLanguages.includes(language)) {
      setCurrentLanguage(language);
      localStorage.setItem('user_language_preference', language);
    }
  };
  
  // 自动检测语言
  const detectLanguage = () => {
    const savedLanguage = localStorage.getItem('user_language_preference') as 'zh' | 'en';
    if (savedLanguage && config.supportedLanguages.includes(savedLanguage)) {
      return savedLanguage;
    }
    
    const browserLanguage = navigator.language || navigator.languages[0];
    if (browserLanguage.startsWith('zh')) {
      return 'zh';
    }
    
    return config.defaultLanguage;
  };
  
  // 初始化语言设置
  useEffect(() => {
    const initialLanguage = detectLanguage();
    setCurrentLanguage(initialLanguage);
  }, []);
  
  return {
    currentLanguage,
    t,
    switchLanguage,
    isLoading,
    resources,
    supportedLanguages: config.supportedLanguages
  };
};
```

### 语言资源加载器
```typescript
// utils/i18nLoader.ts
const languageCache = new Map<string, any>();

export const loadLanguageResources = async (language: 'zh' | 'en') => {
  // 检查缓存
  if (languageCache.has(language)) {
    return languageCache.get(language);
  }
  
  try {
    // 动态导入语言资源
    const [
      cart,
      fields,
      products,
      tooltips,
      validation
    ] = await Promise.all([
      import(`../locales/${language}/cart.json`),
      import(`../locales/${language}/fields.json`),
      import(`../locales/${language}/products.json`),
      import(`../locales/${language}/tooltips.json`),
      import(`../locales/${language}/validation.json`)
    ]);
    
    const resources = {
      cart: cart.default,
      fields: fields.default,
      products: products.default,
      tooltips: tooltips.default,
      validation: validation.default
    };
    
    // 缓存资源
    languageCache.set(language, resources);
    
    return resources;
  } catch (error) {
    console.error(`加载语言资源失败: ${language}`, error);
    throw error;
  }
};

// 清理缓存
export const clearLanguageCache = () => {
  languageCache.clear();
};

// 预加载语言资源
export const preloadLanguageResources = async (languages: ('zh' | 'en')[]) => {
  await Promise.allSettled(
    languages.map(lang => loadLanguageResources(lang))
  );
};
```

```typescript
import { useI18n } from '../hooks/useI18n';

export const useCartFieldDisplayI18n = ({
  productType,
  scenario
}: UseCartFieldDisplayProps) => {
  const { t, currentLanguage } = useI18n();
  const { formatFieldsForCart } = useCartFieldDisplay({ productType, scenario });
  
  // 增强字段标签获取
  const getI18nFieldLabel = useMemo(() => {
    return (fieldKey: string) => {
      const labelKey = `fields.${productType}.${fieldKey}`;
      const label = t(labelKey);
      
      // 如果没有找到翻译，使用原字段名
      if (label === labelKey) {
        // 从字段配置中获取显示名称
        const fieldConfig = fieldConfigs[fieldKey];
        if (fieldConfig?.displayName) {
          return fieldConfig.displayName[currentLanguage] || fieldConfig.displayName.zh || fieldKey;
        }
        return fieldKey;
      }
      
      return label;
    };
  }, [t, productType, currentLanguage]);
  
  // 增强字段格式化
  const formatI18nFieldsForCart = useMemo(() => {
    return (product: any, maxFields?: number) => {
      const fields = formatFieldsForCart(product, maxFields);
      
      return fields.map(field => ({
        ...field,
        label: getI18nFieldLabel(field.key),
        i18nKey: `fields.${productType}.${field.key}`
      }));
    };
  }, [formatFieldsForCart, getI18nFieldLabel, productType]);
  
  return {
    getI18nFieldLabel,
    formatI18nFieldsForCart,
    currentLanguage
  };
};
```

```typescript
export const useResponsiveI18n = () => {
  const { t, currentLanguage } = useI18n();
  const [screenSize, setScreenSize] = useState<'mobile' | 'tablet' | 'desktop'>('desktop');
  
  useEffect(() => {
    const updateScreenSize = () => {
      const width = window.innerWidth;
      if (width < 768) {
        setScreenSize('mobile');
      } else if (width < 1024) {
        setScreenSize('tablet');
      } else {
        setScreenSize('desktop');
      }
    };
    
    updateScreenSize();
    window.addEventListener('resize', updateScreenSize);
    return () => window.removeEventListener('resize', updateScreenSize);
  }, []);
  
  const responsiveT = (key: string, params?: any, fallback?: string) => {
    // 移动端优先使用简短文案
    if (screenSize === 'mobile') {
      const mobileKey = `mobile.${currentLanguage}.${key.split('.').pop()}_short`;
      const mobileText = t(mobileKey);
      if (mobileText !== mobileKey) {
        return mobileText;
      }
    }
    
    return t(key, params, fallback);
  };
  
  return {
    t: responsiveT,
    screenSize,
    currentLanguage
  };
};
```
