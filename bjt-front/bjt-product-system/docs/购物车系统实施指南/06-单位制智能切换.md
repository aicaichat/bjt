# 单位制智能切换实现方案

## 🎯 智能单位制目标

基于现有的 `useAuth().getPreferredUnit()` 用户偏好设置，实现购物车系统中重量、尺寸、长度等字段的智能公制/英制切换，确保全球用户的最佳体验。

## 🔧 基于现有认证系统的智能单位制

### 核心实现策略
利用现有的 `useAuth().getPreferredUnit()` 接口，无需重复实现地区检测，直接基于用户已设置的单位制偏好进行智能字段映射。

```typescript
// 基于现有认证系统的智能单位制Hook
import { useAuth } from '../contexts/AuthContext';

export const useSmartUnitSystem = () => {
  const { getPreferredUnit } = useAuth();
  const [manualOverride, setManualOverride] = useState<'metric' | 'imperial' | null>(null);
  
  // 获取最终的单位制偏好（优先级：手动设置 > 用户偏好 > 默认公制）
  const preferredUnitSystem = useMemo((): 'metric' | 'imperial' => {
    // 1. 手动临时设置优先级最高
    if (manualOverride) {
      return manualOverride;
    }
    
    // 2. 用户账户设置（通过现有认证系统）
    const authPreference = getPreferredUnit?.(); // 'metric' | 'imperial'
    if (authPreference && ['metric', 'imperial'].includes(authPreference)) {
      return authPreference as 'metric' | 'imperial';
    }
    
    // 3. 默认值：全球默认使用公制
    return 'metric';
  }, [manualOverride, getPreferredUnit]);
  
  // 手动临时切换单位制（不影响账户设置）
  const toggleUnitSystem = useCallback(() => {
    const newSystem = preferredUnitSystem === 'metric' ? 'imperial' : 'metric';
    setManualOverride(newSystem);
    
    // 保存到会话存储（临时设置）
    sessionStorage.setItem('temp_unit_override', newSystem);
  }, [preferredUnitSystem]);
  
  // 重置为账户设置
  const resetToAccountSetting = useCallback(() => {
    setManualOverride(null);
    sessionStorage.removeItem('temp_unit_override');
  }, []);
  
  // 初始化临时设置
  useEffect(() => {
    const savedOverride = sessionStorage.getItem('temp_unit_override');
    if (savedOverride && ['metric', 'imperial'].includes(savedOverride)) {
      setManualOverride(savedOverride as 'metric' | 'imperial');
    }
  }, []);
  
  return {
    preferredUnitSystem,
    toggleUnitSystem,
    resetToAccountSetting,
    isTemporaryOverride: !!manualOverride,
    accountUnitSetting: getPreferredUnit?.() || 'metric'
  };
};
```

## 🎯 字段智能映射实现

### 智能字段选择器（基于机器页面成功经验）
```typescript
export const useSmartFieldMapping = () => {
  const { preferredUnitSystem } = useSmartUnitSystem();
  
  // 根据用户偏好自动选择字段（借鉴机器页面实现）
  const getSmartFieldMapping = useCallback((fieldKey: string, product: any) => {
    const fieldConfig = CART_FIELD_CONFIGS[fieldKey];
    if (!fieldConfig?.unitConfig) return fieldKey;
    
    const isImperial = preferredUnitSystem === 'imperial';
    const targetField = isImperial ? 
      fieldConfig.unitConfig.imperial : 
      fieldConfig.unitConfig.metric;
    
    // 检查目标字段是否存在且有值
    if (product[targetField] !== undefined && 
        product[targetField] !== null &&
        product[targetField] !== '') {
      return targetField;
    }
    
    // 如果目标字段不存在，尝试另一个单位制的字段
    const fallbackField = isImperial ? 
      fieldConfig.unitConfig.metric : 
      fieldConfig.unitConfig.imperial;
    
    if (product[fallbackField] !== undefined && 
        product[fallbackField] !== null &&
        product[fallbackField] !== '') {
      return fallbackField;
    }
    
    // 都不存在则返回基础字段名
    return fieldKey;
  }, [preferredUnitSystem]);
  
  return { getSmartFieldMapping, preferredUnitSystem };
};
```

## 🔄 单位制转换规则

### 重量转换
```typescript
export const WEIGHT_CONVERSIONS = {
  // 公制到英制
  kg_to_lbs: (kg: number): number => kg * 2.20462,
  g_to_oz: (g: number): number => g * 0.035274,
  
  // 英制到公制
  lbs_to_kg: (lbs: number): number => lbs / 2.20462,
  oz_to_g: (oz: number): number => oz / 0.035274,
  
  // 格式化
  formatWeight: (value: number, unit: string): string => {
    if (unit === 'kg' || unit === 'lbs') {
      return value.toFixed(2);
    }
    return value.toFixed(1);
  }
};
```

### 尺寸转换
```typescript
export const DIMENSION_CONVERSIONS = {
  // 长度转换
  cm_to_inch: (cm: number): number => cm / 2.54,
  inch_to_cm: (inch: number): number => inch * 2.54,
  m_to_ft: (m: number): number => m * 3.28084,
  ft_to_m: (ft: number): number => ft / 3.28084,
  
  // 复合尺寸转换
  convertCompositeDimension: (
    value: string, 
    fromUnit: 'cm' | 'inch'
  ): { value: string; unit: 'cm' | 'inch' } => {
    if (!value || typeof value !== 'string') {
      return { value: '', unit: fromUnit };
    }
    
    // 解析复合尺寸（如 "75*35*45" 或 "29.5*13.8*17.7"）
    const dimensions = value.split(/[*×x]/).map(d => parseFloat(d.trim()));
    if (dimensions.some(isNaN)) {
      return { value, unit: fromUnit };
    }
    
    const targetUnit = fromUnit === 'cm' ? 'inch' : 'cm';
    const converter = fromUnit === 'cm' ? 
      DIMENSION_CONVERSIONS.cm_to_inch : 
      DIMENSION_CONVERSIONS.inch_to_cm;
    
    const convertedDimensions = dimensions.map(d => 
      Number(converter(d).toFixed(1))
    );
    
    return {
      value: convertedDimensions.join('*'),
      unit: targetUnit
    };
  },
  
  // 格式化
  formatDimension: (value: number, unit: string): string => {
    return value.toFixed(1);
  }
};
```

```typescript
export const useSmartFieldLabels = () => {
  const { preferredUnitSystem } = useSmartUnitSystem();
  const { currentLanguage } = useI18n();
  
  const generateUnitLabel = useCallback((
    fieldType: 'weight' | 'dimension' | 'length' | 'diameter',
    customUnit?: string
  ): string => {
    if (customUnit) return customUnit;
    
    const unitLabels = {
      zh: {
        weight: {
          metric: 'kg',
          imperial: 'lbs'
        },
        dimension: {
          metric: 'cm',
          imperial: 'inch'
        },
        length: {
          metric: 'm',
          imperial: 'ft'
        },
        diameter: {
          metric: 'cm',
          imperial: 'inch'
        }
      },
      en: {
        weight: {
          metric: 'kg',
          imperial: 'lbs'
        },
        dimension: {
          metric: 'cm',
          imperial: 'inch'
        },
        length: {
          metric: 'm',
          imperial: 'ft'
        },
        diameter: {
          metric: 'cm',
          imperial: 'inch'
        }
      }
    };
    
    return unitLabels[currentLanguage]?.[fieldType]?.[preferredUnitSystem] || '';
  }, [preferredUnitSystem, currentLanguage]);
  
  const generateFieldLabel = useCallback((
    baseLabel: string,
    fieldType?: 'weight' | 'dimension' | 'length' | 'diameter',
    customUnit?: string
  ): string => {
    if (!fieldType) return baseLabel;
    
    const unit = generateUnitLabel(fieldType, customUnit);
    if (!unit) return baseLabel;
    
    // 避免重复单位
    if (baseLabel.includes(`(${unit})`)) {
      return baseLabel;
    }
    
    return `${baseLabel}(${unit})`;
  }, [generateUnitLabel]);
  
  return {
    generateUnitLabel,
    generateFieldLabel,
    preferredUnitSystem,
    currentLanguage
  };
};
```

### SmartFieldValue 组件
```typescript
export const SmartFieldValue: React.FC<{
  product: any;
  fieldKey: string;
  enableConversion?: boolean;
  showUnit?: boolean;
  precision?: number;
}> = ({ 
  product, 
  fieldKey, 
  enableConversion = false,
  showUnit = false,
  precision = 2
}) => {
  const { getSmartFieldMapping } = useSmartFieldMapping();
  
  const targetField = getSmartFieldMapping(fieldKey, product);
  const rawValue = product[targetField];
  
  // 如果没有值，返回空
  if (rawValue === undefined || rawValue === null || rawValue === '') {
    return <span className="text-gray-400">-</span>;
  }
  
  // 直接显示原值（数据库中已包含正确的单位制数据）
  const displayValue = typeof rawValue === 'number' ? 
    rawValue.toFixed(precision) : 
    rawValue;
  
  return (
    <span className="smart-field-value" data-field={targetField}>
      {displayValue}
    </span>
  );
};
```
