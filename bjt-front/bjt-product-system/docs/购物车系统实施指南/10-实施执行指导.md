## 🔧 Phase 2: 核心Hook开发（2小时）

### Step 2.1: 创建智能单位制Hook
```typescript
// frontend/src/hooks/useSmartUnitSystem.ts
import { useMemo, useState } from 'react';
import { useAuth } from '../contexts/AuthContext';

export const useSmartUnitSystem = () => {
  const { getPreferredUnit } = useAuth();
  const [manualOverride, setManualOverride] = useState<'metric' | 'imperial' | null>(null);
  
  // 获取最终的单位制偏好
  const preferredUnitSystem = useMemo((): 'metric' | 'imperial' => {
    // 1. 手动临时设置优先级最高
    if (manualOverride) {
      return manualOverride;
    }
    
    // 2. 用户账户设置
    const authPreference = getPreferredUnit?.();
    if (authPreference && ['metric', 'imperial'].includes(authPreference)) {
      return authPreference as 'metric' | 'imperial';
    }
    
    // 3. 默认公制
    return 'metric';
  }, [manualOverride, getPreferredUnit]);
  
  // 智能字段选择
  const getSmartFieldKey = (baseField: string): string => {
    const fieldMappings: Record<string, Record<string, string>> = {
      'package_size': {
        metric: 'package_size_cm',
        imperial: 'package_size_inch'
      },
      'net_weight': {
        metric: 'net_weight_kg',
        imperial: 'net_weight_lbs'
      },
      'pallet_size': {
        metric: 'pallet_size_cm',
        imperial: 'pallet_size_inch'
      },
      'bubble_diameter': {
        metric: 'bubble_diameter_cm',
        imperial: 'bubble_diameter_inch'
      },
      'model': {
        metric: 'model_metric',
        imperial: 'model_imperial'
      },
      'spec': {
        metric: 'spec',
        imperial: 'spec_imperial'
      }
    };
    
    const mapping = fieldMappings[baseField];
    return mapping ? mapping[preferredUnitSystem] : baseField;
  };
  
  // 临时切换单位制
  const setTemporaryUnit = (unit: 'metric' | 'imperial' | null) => {
    setManualOverride(unit);
    // 存储到sessionStorage，页面刷新后保持
    if (unit) {
      sessionStorage.setItem('temp_unit_override', unit);
    } else {
      sessionStorage.removeItem('temp_unit_override');
    }
  };
  
  // 初始化时恢复临时设置
  React.useEffect(() => {
    const tempUnit = sessionStorage.getItem('temp_unit_override');
    if (tempUnit && ['metric', 'imperial'].includes(tempUnit)) {
      setManualOverride(tempUnit as 'metric' | 'imperial');
    }
  }, []);
  
  return {
    preferredUnitSystem,
    getSmartFieldKey,
    setTemporaryUnit,
    isTemporaryOverride: !!manualOverride
  };
};
```

### Step 2.2: 创建购物车数据适配Hook
```typescript
// frontend/src/hooks/useCartDisplayEnhancer.ts
import { useMemo } from 'react';
import { useSmartUnitSystem } from './useSmartUnitSystem';
import { useI18n } from 'react-i18next';

export const useCartDisplayEnhancer = (originalData: any, productType: string) => {
  const { preferredUnitSystem, getSmartFieldKey } = useSmartUnitSystem();
  const { i18n } = useI18n();
  
  // 产品类型字段配置
  const fieldConfigs = useMemo(() => ({
    machines: {
      cartFields: ['model', 'voltage', 'image_url', 'part_number', 'name', 'pcs_per_box', 'pallet_size', 'pcs_per_pallet'],
      smartFields: ['pallet_size']
    },
    consumables: {
      cartFields: ['app_model', 'name_en', 'image_url', 'part_number', 'model', 'spec', 'bubble_diameter', 'product_id', 'pcs_per_box'],
      smartFields: ['model', 'spec', 'bubble_diameter']
    },
    spareParts: {
      cartFields: ['app_model', 'image_url', 'part_number', 'name_en', 'spec', 'app_sn', 'package_size', 'unit', 'net_weight', 'pcs_per_box'],
      smartFields: ['package_size', 'net_weight']
    },
    accessories: {
      cartFields: ['image_url', 'model', 'part_number', 'product_name', 'voltage_v', 'frequency_hz', 'pcs_per_box'],
      smartFields: []
    }
  }), []);
  
  // 生成增强后的显示数据
  const enhancedData = useMemo(() => {
    if (!originalData) return null;
    
    const config = fieldConfigs[productType];
    if (!config) return originalData;
    
    const enhanced = { ...originalData };
    
    // 添加智能单位制字段的显示数据
    enhanced._display = {};
    
    config.smartFields.forEach(baseField => {
      const smartKey = getSmartFieldKey(baseField);
      const value = originalData[smartKey];
      
      if (value !== undefined && value !== null) {
        enhanced._display[baseField] = {
          value,
          unit: getUnitForField(baseField, preferredUnitSystem),
          formatted: formatFieldValue(value, baseField, preferredUnitSystem)
        };
      }
    });
    
    // 添加字段标签的多语言显示
    enhanced._labels = {};
    config.cartFields.forEach(field => {
      enhanced._labels[field] = getFieldLabel(field, productType, i18n.language);
    });
    
    return enhanced;
  }, [originalData, productType, preferredUnitSystem, i18n.language]);
  
  return enhancedData;
};

// 辅助函数
const getUnitForField = (field: string, unitSystem: 'metric' | 'imperial'): string => {
  const units = {
    package_size: { metric: 'cm', imperial: 'inch' },
    net_weight: { metric: 'kg', imperial: 'lb' },
    pallet_size: { metric: 'cm', imperial: 'inch' },
    bubble_diameter: { metric: 'mm', imperial: 'inch' }
  };
  
  return units[field]?.[unitSystem] || '';
};

const formatFieldValue = (value: any, field: string, unitSystem: 'metric' | 'imperial'): string => {
  if (typeof value === 'number') {
    return value.toFixed(field.includes('weight') ? 2 : 1);
  }
  return String(value || '');
};

const getFieldLabel = (field: string, productType: string, language: string): { zh: string; en: string } => {
  // 基于CSV标准的字段标签映射
  const labels = {
    model: { zh: '型号', en: 'Model' },
    voltage: { zh: '电压', en: 'Voltage' },
    image_url: { zh: '产品图片', en: 'Product Image' },
    part_number: { zh: '料号', en: 'Part No.' },
    name: { zh: '名称', en: 'Item' },
    pcs_per_box: { zh: '单箱数量', en: 'Qty per Carton' },
    pallet_size: { zh: '托盘尺寸', en: 'Pallet Size' },
    net_weight: { zh: '单件净重', en: 'Net Weight' },
    unit: { zh: '单位', en: 'Unit' }
  };
  
  return labels[field] || { zh: field, en: field };
};
```

```typescript
// frontend/src/components/SmartAddToCartButton.tsx
import React from 'react';
import { useCartDisplayEnhancer } from '../hooks/useCartDisplayEnhancer';

interface SmartAddToCartButtonProps {
  product: any;
  productType: 'machines' | 'consumables' | 'spareParts' | 'accessories';
  onAddToCart: (product: any) => void;
  children?: React.ReactNode;
  className?: string;
}

export const SmartAddToCartButton: React.FC<SmartAddToCartButtonProps> = ({
  product,
  productType,
  onAddToCart,
  children,
  className = ''
}) => {
  const enhancedProduct = useCartDisplayEnhancer(product, productType);
  
  const handleClick = () => {
    // 重要：传递原始数据给现有的处理函数，保持兼容性
    onAddToCart(product);
  };
  
  return (
    <button 
      onClick={handleClick}
      className={`add-to-cart-btn ${className}`}
      data-product-type={productType}
      data-enhanced-fields={JSON.stringify(enhancedProduct?._display || {})}
    >
      {children || '添加到购物车'}
    </button>
  );
};
```

### Step 3.2: 升级侧边栏购物车
```typescript
// frontend/src/components/SmartSidebarCart.tsx
import React from 'react';
import { useCartDisplayEnhancer } from '../hooks/useCartDisplayEnhancer';
import { useSmartUnitSystem } from '../hooks/useSmartUnitSystem';

interface CartItem {
  id: string;
  product: any;
  productType: string;
  quantity: number;
}

interface SmartSidebarCartProps {
  cartItems: CartItem[];
  onUpdateQuantity: (id: string, quantity: number) => void;
  onRemoveItem: (id: string) => void;
  onCheckout: () => void;
}

export const SmartSidebarCart: React.FC<SmartSidebarCartProps> = ({
  cartItems,
  onUpdateQuantity,
  onRemoveItem,
  onCheckout
}) => {
  const { preferredUnitSystem, setTemporaryUnit, isTemporaryOverride } = useSmartUnitSystem();
  
  return (
    <div className="smart-sidebar-cart">
      {/* 单位制切换器 */}
      <div className="unit-system-toggle">
        <button 
          onClick={() => setTemporaryUnit(preferredUnitSystem === 'metric' ? 'imperial' : 'metric')}
          className="unit-toggle-btn"
        >
          {preferredUnitSystem === 'metric' ? '切换到英制' : '切换到公制'}
        </button>
        {isTemporaryOverride && (
          <button 
            onClick={() => setTemporaryUnit(null)}
            className="reset-unit-btn"
          >
            恢复账户设置
          </button>
        )}
      </div>
      
      {/* 购物车商品列表 */}
      {cartItems.map(item => (
        <SmartCartItem
          key={item.id}
          item={item}
          onUpdateQuantity={onUpdateQuantity}
          onRemoveItem={onRemoveItem}
        />
      ))}
      
      {/* 结算按钮 */}
      <button onClick={onCheckout} className="checkout-btn">
        去结算
      </button>
    </div>
  );
};

const SmartCartItem: React.FC<{
  item: CartItem;
  onUpdateQuantity: (id: string, quantity: number) => void;
  onRemoveItem: (id: string) => void;
}> = ({ item, onUpdateQuantity, onRemoveItem }) => {
  const enhancedProduct = useCartDisplayEnhancer(item.product, item.productType);
  
  return (
    <div className="smart-cart-item">
      {/* 产品图片 */}
      <img src={enhancedProduct?.image_url} alt={enhancedProduct?.name} />
      
      {/* 产品信息 - 使用增强后的显示数据 */}
      <div className="product-info">
        <h4>{enhancedProduct?.name}</h4>
        <p>料号: {enhancedProduct?.part_number}</p>
        
        {/* 智能单位制字段显示 */}
        {enhancedProduct?._display && Object.entries(enhancedProduct._display).map(([field, data]: [string, any]) => (
          <p key={field}>
            {enhancedProduct._labels[field]?.zh}: {data.formatted} {data.unit}
          </p>
        ))}
      </div>
      
      {/* 数量控制 */}
      <div className="quantity-controls">
        <button onClick={() => onUpdateQuantity(item.id, item.quantity - 1)}>-</button>
        <span>{item.quantity}</span>
        <button onClick={() => onUpdateQuantity(item.id, item.quantity + 1)}>+</button>
      </div>
      
      {/* 删除按钮 */}
      <button onClick={() => onRemoveItem(item.id)}>删除</button>
    </div>
  );
};
```
