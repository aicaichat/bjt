### 智能单位制切换 🔄
```typescript
// 优先级系统：手动临时设置 > 账户偏好 > 默认公制
const preferredUnitSystem = useMemo(() => {
  if (manualOverride) return manualOverride;          // 1. 临时设置
  if (authPreference) return authPreference;          // 2. 账户设置
  return 'metric';                                    // 3. 默认公制
}, [manualOverride, getPreferredUnit]);
```

### 字段智能映射 🎯
```typescript
// 基于CSV标准的51个字段智能映射
const getSmartFieldKey = (baseField: string): string => {
  const fieldMappings = {
    'package_size': { metric: 'package_size_cm', imperial: 'package_size_inch' },
    'net_weight': { metric: 'net_weight_kg', imperial: 'net_weight_lbs' },
    'bubble_diameter': { metric: 'bubble_diameter_cm', imperial: 'bubble_diameter_inch' }
    // ... 48 more field mappings
  };
  return fieldMappings[baseField]?.[preferredUnitSystem] || baseField;
};
```
