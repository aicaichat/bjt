智能语言检测算法
function isChineseText(text: string): boolean {
  const chineseMatches = text.match(/[\u4e00-\u9fff]/g);
  const chineseRatio = chineseMatches ? chineseMatches.length / totalChars : 0;
  return chineseRatio > 0.3; // 中文字符占比超过30%
}
